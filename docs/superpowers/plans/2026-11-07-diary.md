# Denník Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A daily diary (one entry per day, two fields) with an offline check of the Spanish words, a Domov card,
sync with the account and the JSON backup.

**Architecture:** A new Dexie table `diary` keyed by the local date; it joins the record-by-record sync as table kind
`diary` (the generic outbox hooks, adoption and sign-out wipe pick it up through `TABLE_KINDS`). A pure module
`diaryCheck.ts` splits the Spanish text and looks every word up in the known forms. Three screens under
`src/features/diary/`.

**Tech Stack:** React + TypeScript, Dexie (`dexie-react-hooks`), vitest + fake-indexeddb, Playwright.

**Spec:** `docs/superpowers/specs/2026-11-07-diary-design.md`

## Global Constraints

- UI strings in Slovak (`src/i18n/sk.ts`) and English (`src/i18n/en.ts`); code and comments in English.
- Spanish content Mexican: no `vosotros`, no Spain-only words (checked by `validate:data`).
- Each field ≤ 5000 characters.
- "Today" = the device's local date, `dayKey(new Date())` from `src/lib/dates.ts`.
- Diary does not count toward the daily goal or the streak (no `recordAttempt`).
- Mobile first 375 px, tap targets ≥ 44 px.
- Work in a worktree `C:\CODE\SOMOS-diary`, branch `diary`; e2e with `E2E_PORT=5299`.
- Stage explicit paths; never `CLAUDE.md`, `poznaky.txt`, `.gitignore`, `Design/`.
- Push only after Adam ran the SQL of Task 1 in Supabase and said so.

## Review Focus

1. Typing fast then leaving the page at once (back arrow, tab bar): the last characters must be saved (flush on unmount).
2. Opening `/diary/2026-13-45`, `/diary/abc` or tomorrow's date: back to `/diary`, nothing stored.
3. An entry edited on two devices: the newer one wins on both after a sync; a day cleared on one device disappears on the other.
4. A Spanish text with punctuation, ¿¡, numbers, names and line breaks: rendered exactly as typed; only unknown words underlined.
5. A backup from before the diary existed restores without errors; a backup with a broken diary row skips only that row.

```ts
import { useLiveQuery } from 'dexie-react-hooks'
import { dayKey } from './dates'
import { db, type DiaryEntry } from './db'

// Denník: one entry per local day. Saved while typing; both fields empty = the day is removed.
// Synced as table kind "diary" (src/lib/sync/kinds.ts) and carried by the backup.

export const DIARY_MAX = 5000
const DATE = /^(\d{4})-(\d{2})-(\d{2})$/

export const todayKey = () => dayKey(new Date())

/** A real calendar day, not after `today`. */
export function isDiaryDate(date: string, today = todayKey()): boolean {
  const m = DATE.exec(date)
  if (!m) return false
  const d = new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]))
  return dayKey(d) === date && date <= today
}

const isStr = (x: unknown): x is string => typeof x === 'string'
const isNum = (x: unknown): x is number => typeof x === 'number' && Number.isFinite(x)

export function isDiaryEntry(x: unknown): x is DiaryEntry {
  if (typeof x !== 'object' || x === null) return false
  const e = x as Record<string, unknown>
  return isStr(e.date) && DATE.test(e.date) && isStr(e.learned) && isStr(e.es) && isNum(e.createdAt) && isNum(e.updatedAt)
}

export async function saveDiaryEntry(date: string, fields: { learned: string; es: string }, now = Date.now()): Promise<'saved' | 'removed' | 'unchanged'> {
  const learned = fields.learned.slice(0, DIARY_MAX)
  const es = fields.es.slice(0, DIARY_MAX)
  return db.transaction('rw', db.diary, async () => {
    const existing = await db.diary.get(date)
    if (!learned.trim() && !es.trim()) {
      if (!existing) return 'unchanged'
      await db.diary.delete(date)
      return 'removed'
    }
    if (existing && existing.learned === learned && existing.es === es) return 'unchanged'
    await db.diary.put({ date, learned, es, createdAt: existing?.createdAt ?? now, updatedAt: now })
    return 'saved'
  })
}

/** The rows of a backup that should be written here: days not here, or edited later than here. */
export function newerDiaryEntries(local: DiaryEntry[], incoming: DiaryEntry[]): DiaryEntry[] {
  const here = new Map(local.map((e) => [e.date, e.updatedAt]))
  return incoming.filter((e) => e.updatedAt > (here.get(e.date) ?? -Infinity))
}

/** `null` = no entry that day, `undefined` = still loading. */
export function useDiaryEntry(date: string): DiaryEntry | null | undefined {
  return useLiveQuery(async () => (await db.diary.get(date)) ?? null, [date])
}

/** Every entry, newest day first. */
export function useDiaryEntries(): DiaryEntry[] | undefined {
  return useLiveQuery(() => db.diary.orderBy('date').reverse().toArray(), [])
}
```

```ts
import { words, wordIdByVerb } from '../data'
import { lookupForm, lookupVerb } from './knownForms'
import { fold } from './text'

// The check under "Po španielsky" in the diary: every word looked up in the dataset's forms and the learner's
// own words. Offline and cheap; a real word in the wrong place (esta for está) is not caught, only grammar could.

export type DiarySegment = { text: string } | { text: string; word: true; known: boolean; suggestions: string[]; href?: string }

const WORD = /\p{L}+(?:['’-]\p{L}+)*/gu
const SENTENCE_END = /[.!?¡¿\n]\s*$/

let folded: Map<string, string[]> | undefined
/** Known forms by their spelling without accents and ñ: "manana" → ["mañana"]. */
function foldedIndex(): Map<string, string[]> {
  if (folded) return folded
  folded = new Map()
  for (const w of words) for (const form of [w.es, w.plural, w.feminine]) {
    if (!form || form.includes(' ')) continue
    const key = fold(form)
    const list = folded.get(key) ?? []
    if (!list.includes(form)) list.push(form)
    folded.set(key, list)
  }
  return folded
}

const wordIdByEs = new Map(words.map((w) => [w.es.toLowerCase(), w.id]))

export function checkDiary(text: string, customWords: readonly string[] = []): DiarySegment[] {
  const custom = new Set(customWords.map((w) => w.toLowerCase()))
  const out: DiarySegment[] = []
  let at = 0
  for (const match of text.matchAll(WORD)) {
    const start = match.index
    if (start > at) out.push({ text: text.slice(at, start) })
    const word = match[0]
    const lower = word.toLowerCase()
    const sentenceStart = start === 0 || SENTENCE_END.test(text.slice(0, start))
    const name = !sentenceStart && word[0] !== word[0].toLowerCase()
    const known = name || custom.has(lower) || lookupForm(lower) !== undefined
    const verb = lookupVerb(lower)
    const id = wordIdByEs.get(lower) ?? (verb ? wordIdByVerb.get(verb) : undefined)
    out.push({
      text: word,
      word: true,
      known,
      suggestions: known ? [] : (foldedIndex().get(fold(word)) ?? []).slice(0, 2),
      href: known && id ? `/word/${id}` : undefined,
    })
    at = start + word.length
  }
  if (at < text.length) out.push({ text: text.slice(at) })
  return out
}
```
Numbers never match `WORD` (letters only), so they stay plain text. If `foldedIndex` misses a verb form suggestion such
as *esta* → *está* that is fine (both are real words; by design).

```ts
export function promptOfDay(date: string): DiaryPrompt {
  const [y, m, d] = date.split('-').map(Number)
  const dayOfYear = Math.round((Date.UTC(y, m - 1, d) - Date.UTC(y, 0, 1)) / 86_400_000)
  return diaryPrompts[dayOfYear % diaryPrompts.length]
}
```
