> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make "Viem to" a visible button, offer another start level once after the first lesson of the path, and celebrate a finished level.

**Architecture:** Pure decisions in `src/lib/path.ts` (`isLevelOfferLesson`, `levelOffer`, `levelDone`, `levelSummary`), the onboarding state learns `levelOffered` and B2. Two new cards in `src/features/path/` (`LevelOffer`, `LevelDone`) that mirror the existing `SaveProgressCard` and `UnitDone`; `LessonPage`, `TestPage`/`TestResult` and `PathPage` only mount them.

**Tech Stack:** Vite + React + TypeScript, Tailwind tokens, react-router, Vitest, Playwright.

**Spec:** `docs/guided-path-spec.md` §10c (read it first).

## Global Constraints

- UI strings in Slovak (`src/i18n/sk.ts`) and English (`src/i18n/en.ts`, typed `Dictionary`); code, comments, commits in English.
- No new server tables, no sync change: the onboarding state stays on the device (`localStorage` key `somos-onboarding`).
- The welcome's level choice stays `START_LEVELS = ['A1', 'A2', 'B1']`; only the stored level also accepts `B2`.
- Offer thresholds: up when `score === total`; down when `score <= total * 0.4`; only in the first full round; once per device (`levelOffered`).
- Mobile first, 375 px, tap targets ≥ 44 px (`min-h-11` / `h-12`).
- Work in a worktree on branch `path-phase-3` from `main`; never stage `CLAUDE.md`, `poznaky.txt`, `.gitignore`, `Design/`; stage explicit paths; no push or merge without Adam's word.
- e2e in the worktree: `E2E_PORT=5299 npm run test:e2e -- --workers=3`.

## Review Focus

1. A learner who accepts the offer and then comes back to the first lesson (or repeats it) must never see the offer again → `levelOffered` is stored when the result opens; unit test in Task 2 ("once offered"), and the e2e of Task 4 plays the A1 lesson again after accepting.
2. A correction round ("Opraviť chyby") or a resumed lesson reaching 10/10 must not count as the first round → `levelOffer(..., firstRound=false)` returns undefined; unit test in Task 2.
3. Repeating an already passed level exam must not celebrate again → `levelWasDone` captured at test start; unit test of `levelDone` in Task 2, the capture reviewed in Task 5.
4. A device without a welcome state (old devices with progress, e.g. Adam's) gets no offer → unit test in Task 2.
5. B2 stored as start level must survive a reload (`parse`) → unit test in Task 1.

```ts
// ---------- level offer and level end (docs/guided-path-spec.md §10c) ----------

export interface LevelOffer {
  direction: 'up' | 'down'
  level: Level
}

/** The lesson the level offer belongs to: step 1 of the first unit of the start level, on a device with a welcome, not offered yet. */
export function isLevelOfferLesson(unit: PathUnit, step: StepId, onboarding: OnboardingState | null): boolean {
  return onboarding !== null && onboarding.levelOffered !== true && firstUnit(onboarding)?.id === unit.id && unitSteps(unit)[0] === step
}

/** All right in the first full round: the level above; at most 40 %: the level below; anything else: nothing. */
export function levelOffer(
  unit: PathUnit,
  step: StepId,
  onboarding: OnboardingState | null,
  firstRound: boolean,
  score: number,
  total: number,
): LevelOffer | undefined {
  if (!firstRound || total <= 0 || !isLevelOfferLesson(unit, step, onboarding)) return undefined
  const at = LEVEL_ORDER.indexOf(startLevelOf(onboarding))
  const up = LEVEL_ORDER[at + 1]
  const down = at > 0 ? LEVEL_ORDER[at - 1] : undefined
  if (score === total && up && pathCovers(up)) return { direction: 'up', level: up }
  if (score <= total * 0.4 && down && pathCovers(down)) return { direction: 'down', level: down }
  return undefined
}

/** Every unit of the level done (its test, the topic test or the level exam). */
export function levelDone(states: readonly UnitState[], level: Level): boolean {
  const units = states.filter((s) => s.unit.level === level)
  return units.length > 0 && units.every((s) => s.done)
}

/** What a level brought, for its celebration: units, words (each taught once) and grammar points. */
export function levelSummary(level: Level): { units: number; words: number; grammar: number } {
  const units = pathUnits.filter((u) => u.level === level)
  return {
    units: units.length,
    words: units.reduce((n, u) => n + unitMaterial(u).words.length, 0),
    grammar: new Set(units.map((u) => u.grammar)).size,
  }
}
```

(`firstUnit`, `startLevelOf`, `pathCovers` are `const` arrow functions defined earlier in the file; they are only called at run time, so order is fine.)

```tsx
import { ArrowDown, ArrowUp } from 'lucide-react'
import { useEffect, useState } from 'react'
import { useNavigate } from 'react-router'
import { Button } from '../../components/Button'
import { useT } from '../../i18n'
import { getOnboarding, updateOnboarding } from '../../lib/onboarding'
import { firstUnit, isLevelOfferLesson, levelOffer, stepHref, unitSteps, type StepId } from '../../lib/path'
import type { PathUnit } from '../../lib/pathUnits'

/**
 * Over the result of the path's first lesson: all right offers the level above, a weak start the one below
 * (docs/guided-path-spec.md §10c). Decided once, when the result opens; from then on it is never offered again.
 */
export function LevelOffer({ unit, step, firstRound, score, total }: { unit: PathUnit; step: StepId; firstRound: boolean; score: number; total: number }) {
  const text = useT().path
  const navigate = useNavigate()
  const [decided] = useState(() => {
    const onboarding = getOnboarding()
    return isLevelOfferLesson(unit, step, onboarding) ? { offer: levelOffer(unit, step, onboarding, firstRound, score, total) } : undefined
  })
  const [open, setOpen] = useState(true)

  useEffect(() => {
    if (decided) updateOnboarding({ levelOffered: true })
  }, [decided])

  const offer = decided?.offer
  if (!offer || !open) return null

  const accept = () => {
    updateOnboarding({ level: offer.level })
    const start = firstUnit()
    if (start) void navigate(stepHref(start, unitSteps(start)[0]), { replace: true })
  }

  const Icon = offer.direction === 'up' ? ArrowUp : ArrowDown
  return (
    <section aria-labelledby="level-offer" className="mb-6 rounded-card border border-brick/40 bg-brick/10 p-5">
      <div className="flex items-start gap-3">
        <Icon size={22} strokeWidth={1.75} className="mt-0.5 shrink-0 text-brick" aria-hidden />
        <div className="min-w-0">
          <h2 id="level-offer" className="font-semibold">
            {offer.direction === 'up' ? text.offerUp(offer.level) : text.offerDown(offer.level)}
          </h2>
          <p className="mt-1 text-sm text-ink-muted">{offer.direction === 'up' ? text.offerUpHint : text.offerDownHint}</p>
        </div>
      </div>
      <div className="mt-4 grid gap-2">
        <Button onClick={accept}>{text.offerYes(offer.level)}</Button>
        <Button variant="secondary" onClick={() => setOpen(false)}>
          {text.offerStay(unit.level)}
        </Button>
      </div>
    </section>
  )
}
```

- [ ] **Step 2: Create `src/features/path/LevelDone.tsx`**

```tsx
import { Award, ArrowRight } from 'lucide-react'
import { useNavigate } from 'react-router'
import { Button } from '../../components/Button'
import type { Level } from '../../data/types'
import { useT } from '../../i18n'
import { levelSummary, nextStep, stepHref, unitTitle, type UnitState } from '../../lib/path'

/**
 * The end of a level, on the result of the test that closed it: a bigger celebration than a unit's,
 * what the level brought and the way on (docs/guided-path-spec.md §10c).
 */
export function LevelDone({ level, states }: { level: Level; states: readonly UnitState[] }) {
  const text = useT().path
  const navigate = useNavigate()
  const next = nextStep(states)
  const { units, words, grammar } = levelSummary(level)

  return (
    <section aria-labelledby="level-done" className="rounded-card border border-leaf/40 bg-leaf/10 px-5 pt-7 pb-5 text-center">
      <span className="mx-auto flex size-20 animate-sun-rise items-center justify-center rounded-full bg-leaf text-on-accent shadow-lg">
        <Award size={40} strokeWidth={1.75} aria-hidden />
      </span>
      <p lang="es" className="mt-4 animate-rise-in font-serif text-3xl font-semibold tracking-tight [animation-delay:200ms]">
        ¡Nivel {level} completado!
      </p>
      <h2 id="level-done" className="mt-1 animate-rise-in text-ink-muted [animation-delay:300ms]">
        {text.levelDoneLine(level)}
      </h2>
      <ul className="mt-3 flex animate-rise-in flex-wrap justify-center gap-2 text-sm [animation-delay:400ms]">
        <li className="rounded-full bg-surface px-3 py-1 font-medium">{text.levelUnits(units)}</li>
        {words > 0 && <li className="rounded-full bg-surface px-3 py-1 font-medium">{text.unitWords(words)}</li>}
        <li className="rounded-full bg-surface px-3 py-1 font-medium">{text.levelGrammar(grammar)}</li>
      </ul>
      {next ? (
        <Button
          icon={ArrowRight}
          onClick={() => void navigate(stepHref(next.unit.unit, next.step.id), { replace: true })}
          className="mt-5 w-full"
        >
          {text.nextUnit(`${next.unit.unit.level} · ${unitTitle(next.unit.unit)}`)}
        </Button>
      ) : (
        <p className="mt-5 font-medium">{text.walked}</p>
      )}
    </section>
  )
}
```
